1. Visão Geral

🎯 Objetivo: Este documento especifica o schema completo do banco de dados PostgreSQL (Supabase) para o sistema TM Gerenciador de O.S.

Tecnologias

  • Database: PostgreSQL 15+ (Supabase)
  • ORM: Prisma (recomendado) ou Drizzle
  • Autenticação: Supabase Auth
  • Storage: Supabase Storage (fotos)
  • Row Level Security: Habilitado

Convenções

  • Nomes de tabelas: snake_case no plural
  • Chaves primárias: id (UUID v4)
  • Chaves estrangeiras: {tabela}_id
  • Timestamps: created_at, updated_at
  • Soft delete: deleted_at (nullable)

Diagrama ER Simplificado

usuarios ──┬─── ordens_servico ──── pacotes_levantamento ──── fotos_levantamento
           │                    └──── status_os
           │
           └─── contratos ──── ordens_servico

2. Tabelas Core (Gerenciador de O.S.)

2.1. usuarios

Armazena informações dos usuários do sistema (Gerentes, Técnicos, Admins). 

CREATE TABLE usuarios (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    nome VARCHAR(255) NOT NULL,
    perfil VARCHAR(50) NOT NULL CHECK (perfil IN (
        'tecnico', 'gerente', 'admin', 'elaborador', 'diretor'
    )),
    telefone VARCHAR(20),
    avatar_url TEXT,
    contrato_id UUID REFERENCES contratos(id),
    ativo BOOLEAN DEFAULT true,
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now(),
    deleted_at TIMESTAMPTZ
);

-- Índices
CREATE INDEX idx_usuarios_email ON usuarios(email);
CREATE INDEX idx_usuarios_perfil ON usuarios(perfil);
CREATE INDEX idx_usuarios_contrato ON usuarios(contrato_id);

Dicionário de Dados

Campo Tipo Constraints Descrição
id PK UUID NOT NULL Identificador único do usuário
email UK VARCHAR(255) UNIQUE, NOT NULL E-mail corporativo (usado para login)
nome Required VARCHAR(255) NOT NULL Nome completo do usuário
perfil VARCHAR(50) CHECK (enum) Perfil de acesso: tecnico, gerente, admin, elaborador, diretor
contrato_id FK UUID REFERENCES contratos Contrato ao qual o usuário pertence

2.2. contratos

CREATE TABLE contratos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    codigo VARCHAR(50) UNIQUE NOT NULL,
    nome VARCHAR(255) NOT NULL,
    cliente VARCHAR(255) NOT NULL,
    data_inicio DATE NOT NULL,
    data_fim DATE,
    status VARCHAR(50) DEFAULT 'ativo' CHECK (status IN (
        'ativo', 'suspenso', 'encerrado'
    )),
    configuracoes JSONB DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX idx_contratos_codigo ON contratos(codigo);
CREATE INDEX idx_contratos_status ON contratos(status);

2.3. ordens_servico

Tabela principal do sistema. Armazena todas as ordens de serviço. 

CREATE TABLE ordens_servico (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    numero_os VARCHAR(50) UNIQUE NOT NULL,
    contrato_id UUID NOT NULL REFERENCES contratos(id),
    tecnico_id UUID REFERENCES usuarios(id),
    
    -- Dados do serviço
    tipo_servico VARCHAR(100) NOT NULL,
    descricao TEXT,
    
    -- Localização
    endereco TEXT NOT NULL,
    cidade VARCHAR(100),
    estado VARCHAR(2),
    cep VARCHAR(10),
    
    -- Dados do cliente
    cliente_nome VARCHAR(255),
    cliente_contato VARCHAR(100),
    
    -- Status e prazos
    status VARCHAR(50) DEFAULT 'criada' CHECK (status IN (
        'criada', 'distribuida', 'em_campo', 
        'levantamento_enviado', 'em_validacao', 
        'validada', 'concluida', 'cancelada'
    )),
    prioridade VARCHAR(20) DEFAULT 'normal' CHECK (prioridade IN (
        'baixa', 'normal', 'alta', 'urgente'
    )),
    prazo DATE,
    data_conclusao TIMESTAMPTZ,
    
    -- Metadados
    criado_por UUID REFERENCES usuarios(id),
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now(),
    deleted_at TIMESTAMPTZ
);

-- Índices para performance
CREATE INDEX idx_os_numero ON ordens_servico(numero_os);
CREATE INDEX idx_os_contrato ON ordens_servico(contrato_id);
CREATE INDEX idx_os_tecnico ON ordens_servico(tecnico_id);
CREATE INDEX idx_os_status ON ordens_servico(status);
CREATE INDEX idx_os_prazo ON ordens_servico(prazo);
CREATE INDEX idx_os_created ON ordens_servico(created_at);

2.4. pacotes_levantamento

CREATE TABLE pacotes_levantamento (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    os_id UUID NOT NULL REFERENCES ordens_servico(id) ON DELETE CASCADE,
    tecnico_id UUID NOT NULL REFERENCES usuarios(id),
    
    -- Dados do pacote
    dados_json JSONB NOT NULL,
    total_fotos INTEGER DEFAULT 0,
    
    -- Status e validação
    status VARCHAR(50) DEFAULT 'pendente' CHECK (status IN (
        'pendente', 'em_validacao', 'validado', 'rejeitado'
    )),
    validado_por UUID REFERENCES usuarios(id),
    validado_em TIMESTAMPTZ,
    feedback_validacao TEXT,
    
    -- Timestamps
    data_coleta TIMESTAMPTZ NOT NULL,
    data_upload TIMESTAMPTZ DEFAULT now(),
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX idx_pacotes_os ON pacotes_levantamento(os_id);
CREATE INDEX idx_pacotes_tecnico ON pacotes_levantamento(tecnico_id);
CREATE INDEX idx_pacotes_status ON pacotes_levantamento(status);

2.5. fotos_levantamento

CREATE TABLE fotos_levantamento (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pacote_id UUID NOT NULL REFERENCES pacotes_levantamento(id) ON DELETE CASCADE,
    
    -- Armazenamento (Supabase Storage)
    storage_path TEXT NOT NULL,
    url_publica TEXT,
    nome_arquivo VARCHAR(255) NOT NULL,
    tamanho_bytes BIGINT,
    
    -- Metadados EXIF
    metadados_exif JSONB,
    latitude DECIMAL(10, 8),
    longitude DECIMAL(11, 8),
    capturado_em TIMESTAMPTZ,
    
    -- Categorização
    categoria VARCHAR(100),
    descricao TEXT,
    
    created_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX idx_fotos_pacote ON fotos_levantamento(pacote_id);
CREATE INDEX idx_fotos_categoria ON fotos_levantamento(categoria);

2.6. status_os (Histórico)

CREATE TABLE status_os (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    os_id UUID NOT NULL REFERENCES ordens_servico(id) ON DELETE CASCADE,
    status_anterior VARCHAR(50),
    status_novo VARCHAR(50) NOT NULL,
    usuario_id UUID REFERENCES usuarios(id),
    observacao TEXT,
    created_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX idx_status_os ON status_os(os_id);
CREATE INDEX idx_status_created ON status_os(created_at);

3. Índices e Otimizações

💡 Performance: Os índices foram criados para otimizar as queries mais comuns do sistema.

Índices Compostos (Queries Complexas)

-- Busca de O.S. por contrato + status
CREATE INDEX idx_os_contrato_status 
ON ordens_servico(contrato_id, status);

-- Busca de O.S. por técnico + status
CREATE INDEX idx_os_tecnico_status 
ON ordens_servico(tecnico_id, status);

-- Busca de O.S. próximas do prazo
CREATE INDEX idx_os_status_prazo 
ON ordens_servico(status, prazo) 
WHERE status NOT IN ('concluida', 'cancelada');

-- Pacotes pendentes de validação
CREATE INDEX idx_pacotes_pendentes 
ON pacotes_levantamento(status, data_upload) 
WHERE status = 'pendente';

Full-Text Search (Opcional)

-- Para busca textual em O.S.
ALTER TABLE ordens_servico 
ADD COLUMN search_vector tsvector;

CREATE INDEX idx_os_search 
ON ordens_servico 
USING gin(search_vector);

-- Trigger para atualizar search_vector
CREATE OR REPLACE FUNCTION update_os_search_vector()
RETURNS TRIGGER AS $$
BEGIN
    NEW.search_vector := 
        to_tsvector('portuguese', 
            coalesce(NEW.numero_os, '') || ' ' ||
            coalesce(NEW.endereco, '') || ' ' ||
            coalesce(NEW.cliente_nome, '')
        );
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_update_os_search
BEFORE INSERT OR UPDATE ON ordens_servico
FOR EACH ROW EXECUTE FUNCTION update_os_search_vector();

4. Políticas RLS (Supabase)

⚠️ Segurança: Row Level Security (RLS) é OBRIGATÓRIO no Supabase para proteger dados.

Habilitar RLS

ALTER TABLE usuarios ENABLE ROW LEVEL SECURITY;
ALTER TABLE contratos ENABLE ROW LEVEL SECURITY;
ALTER TABLE ordens_servico ENABLE ROW LEVEL SECURITY;
ALTER TABLE pacotes_levantamento ENABLE ROW LEVEL SECURITY;
ALTER TABLE fotos_levantamento ENABLE ROW LEVEL SECURITY;
ALTER TABLE status_os ENABLE ROW LEVEL SECURITY;

Políticas para Usuários

-- Usuários podem ver seu próprio perfil
CREATE POLICY "Usuários podem ver próprio perfil"
ON usuarios
FOR SELECT
USING (auth.uid() = id);

-- Gerentes podem ver todos do seu contrato
CREATE POLICY "Gerentes veem usuários do contrato"
ON usuarios
FOR SELECT
USING (
    EXISTS (
        SELECT 1 FROM usuarios u
        WHERE u.id = auth.uid()
        AND u.perfil = 'gerente'
        AND u.contrato_id = usuarios.contrato_id
    )
);

Políticas para Ordens de Serviço

-- Técnicos veem apenas suas O.S.
CREATE POLICY "Técnicos veem suas O.S."
ON ordens_servico
FOR SELECT
USING (
    tecnico_id = auth.uid()
    OR
    EXISTS (
        SELECT 1 FROM usuarios
        WHERE id = auth.uid()
        AND perfil IN ('gerente', 'admin')
    )
);

-- Apenas gerentes podem criar O.S.
CREATE POLICY "Gerentes criam O.S."
ON ordens_servico
FOR INSERT
WITH CHECK (
    EXISTS (
        SELECT 1 FROM usuarios
        WHERE id = auth.uid()
        AND perfil IN ('gerente', 'admin')
    )
);
