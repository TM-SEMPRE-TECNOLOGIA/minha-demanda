Option Explicit

' ==========================================================
'  ORGANIZADOR DE IMAGENS - V1.2
'  Corrige erro de referência (tbl.Cell)
'  Funciona em Word 2010–2024
' ==========================================================
Sub OrganizarImagens_v12()
    Dim doc As Document
    Dim img As InlineShape
    Dim i As Long, total As Long
    Dim grupo() As Range
    Dim grupoCount As Long
    Dim limiteLargura As Single
    Dim espaco As Single

    Set doc = ActiveDocument
    limiteLargura = CentimetersToPoints(7.5)
    espaco = CentimetersToPoints(1.25)

    Application.ScreenUpdating = False
    Application.StatusBar = "Organizando imagens..."

    ReDim grupo(1 To 3)
    grupoCount = 0
    total = doc.InlineShapes.Count

    ' Percorre todas as imagens inline
    For i = 1 To total
        Set img = doc.InlineShapes(i)
        If img.Range.StoryType = wdMainTextStory Then
            grupoCount = grupoCount + 1
            Set grupo(grupoCount) = img.Range.Duplicate
            If grupoCount = 3 Then
                CriarTabelaDeImagens doc, grupo, grupoCount, espaco
                grupoCount = 0
                ReDim grupo(1 To 3)
            End If
        End If
    Next i

    If grupoCount > 0 Then
        CriarTabelaDeImagens doc, grupo, grupoCount, espaco
    End If

    Application.StatusBar = False
    Application.ScreenUpdating = True
    MsgBox "✅ Organização concluída com sucesso (v1.2).", vbInformation
End Sub


' ==========================================================
'  Cria tabela invisível e insere imagens com segurança
' ==========================================================
Private Sub CriarTabelaDeImagens(doc As Document, grupo() As Range, qtd As Long, espaco As Single)
    Dim ins As Range
    Dim tbl As Table
    Dim c As Long
    Dim celula As Range

    ' --- Ponto de inserção isolado ---
    Set ins = grupo(1).Duplicate
    ins.Collapse wdCollapseEnd
    ins.InsertParagraphAfter
    ins.Move wdParagraph, 1 ' move o cursor para fora do parágrafo atual
    Set ins = ins.Paragraphs(1).Range
    ins.InsertParagraphBefore
    ins.Collapse wdCollapseStart

    ' --- Cria tabela invisível ---
    Set tbl = doc.Tables.Add(Range:=ins, NumRows:=1, NumColumns:=qtd)
    With tbl
        .Borders.Enable = False
        .AllowAutoFit = False
        .Rows.Alignment = wdAlignRowCenter
        .Range.ParagraphFormat.Alignment = wdAlignParagraphCenter
        .Range.Cells.VerticalAlignment = wdCellAlignVerticalTop
        If qtd > 1 Then .Spacing = espaco Else .Spacing = 0
    End With

    ' --- Copia imagens para células (FormattedText) ---
    For c = 1 To qtd
        Set celula = tbl.Cell(1, c).Range
        celula.Collapse wdCollapseStart
        celula.FormattedText = grupo(c).FormattedText
        grupo(c).Delete
    Next c

    tbl.Range.InsertParagraphAfter
End Sub
